29def _drive_display_name(original_filename: str | None, extension: str) -> str: 

30 # Cosmetic only (Drive's "name" field, never a filesystem path or lookup 

31 # key — see DriveResumeStore.store_resume) — sanitizing is defence in 

32 # depth against control chars/path separators showing up in Drive UI, 

33 # not a security boundary the way the local storage key is. 

34 stem = Path(original_filename or "").stem.strip() 

35 stem = re.sub(r'[\x00-\x1f/\\:*?"<>|]', "_", stem)[:150].strip() 

36 return f"{stem or 'resume'}.{extension}"

115def submit_application( 

116 db: Session, 

117 job: JobPosting, 

118 form: FormData, 

119 resume: UploadFile, 

120 store: ResumeStore, 

121) -> Candidate: 

122 """The public POST /apply/{slug} orchestration. Order matters and mirrors 

123 the story AC exactly: every check that can reject the submission runs 

124 before anything is stored, the resume is written before the candidate row 

125 is committed (never the reverse), and a storage failure leaves no row.""" 

126 job_service.assert_job_accepting(job) # -> JobNotAccepting, mapped to 410 by main.py 

127 

128 fields_by_id = {field.field_id: field for field in job.template.fields} 

129 

130 answers: dict[uuid.UUID, str] = {} 

131 for key, value in form.multi_items(): 

132 if key == "resume": 

133 continue 

134 if not isinstance(value, str): 

135 raise _unknown_field(key) # an unexpected second file part 

136 try: 

137 field_id = uuid.UUID(key) 

138 except ValueError as exc: 

139 raise _unknown_field(key) from exc 

140 if field_id not in fields_by_id: 

141 raise _unknown_field(key) 

142 answers[field_id] = value.strip() 

143 

144 missing_required = [ 

145 str(field.field_id) 

146 for field in fields_by_id.values() 

147 if field.is_required 

148 and field.field_type != FieldType.FILE_UPLOAD 

149 and not answers.get(field.field_id) 

150 ] 

151 if missing_required: 

152 raise HTTPException( 

153 status_code=422, detail={**_MISSING_REQUIRED, "details": {"missing": missing_required}} 

154 ) 

155 

156 too_long = [str(fid) for fid, value in answers.items() if len(value) > _MAX_RESPONSE_LEN] 

157 if too_long: 

158 raise HTTPException( 

159 status_code=422, 

160 detail={ 

161 "code": "VALIDATION_ERROR", 

162 "message": f"A response exceeds {_MAX_RESPONSE_LEN} characters.", 

163 "details": {"fields": too_long}, 

164 }, 

165 ) 

166 

167 ordered_fields = list(fields_by_id.values()) 

168 email_index, name_index = resolve_identity_fields([f.field_label for f in ordered_fields]) 

169 if email_index is None or name_index is None: 

170 # Backstop only — template_service rejects a template missing either 

171 # field at save time. Reachable only for templates saved before that 

172 # guard existed. 

173 raise HTTPException(status_code=500, detail=_TEMPLATE_MISSING_IDENTITY) 

174 

175 email_raw = answers.get(ordered_fields[email_index].field_id, "") 

176 name_raw = answers.get(ordered_fields[name_index].field_id, "") 

177 if not email_raw or not name_raw: 

178 # Identity fields are required regardless of their is_required flag — 

179 # a NULL here would violate the NOT NULL columns they feed. 

180 raise HTTPException(status_code=422, detail=_MISSING_REQUIRED) 

181 

182 try: 

183 identity = _IdentityAnswers(full_name=name_raw, email=email_raw) 

184 except ValidationError as exc: 

185 raise HTTPException( 

186 status_code=422, 

187 detail={ 

188 "code": "VALIDATION_ERROR", 

189 "message": "Invalid submission data.", 

190 "details": {"errors": exc.errors()}, 

191 }, 

192 ) from exc 

193 

194 existing = db.scalar( 

195 select(Candidate.candidate_id).where( 

196 Candidate.job_id == job.job_id, 

197 Candidate.email == identity.email, 

198 Candidate.deleted_at.is_(None), 

199 ) 

200 ) 

201 if existing is not None: 

202 raise HTTPException(status_code=409, detail=_DUPLICATE_SUBMISSION) 

203 

204 try: 

205 content = read_capped(resume.file, settings.max_resume_mb * 1024 * 1024) 

206 except FileTooLarge as exc: 

207 raise HTTPException( 

208 status_code=413, 

209 detail={ 

210 "code": "FILE_TOO_LARGE", 

211 "message": f"Resume exceeds {settings.max_resume_mb}MB.", 

212 }, 

213 ) from exc 

214 except EmptyFile as exc: 

215 raise HTTPException( 

216 status_code=422, detail={"code": "EMPTY_FILE", "message": "Resume file is empty."} 

217 ) from exc 

218 

219 extension = sniff_extension(content) 

220 if extension is None: 

221 raise HTTPException( 

222 status_code=415, 

223 detail={ 

224 "code": "UNSUPPORTED_FILE_TYPE", 

225 "message": "Only PDF and DOCX resumes are accepted.", 

226 }, 

227 ) 

228 

229 recruiter = db.get(Recruiter, job.recruiter_id) 

230 assert recruiter is not None # FK guarantees this 

231 

232 # Server-generated name only — the candidate's raw filename never reaches 

233 # the filesystem or Drive (TC-08: path traversal / overwrite protection). 

234 stored_filename = f"{uuid.uuid4()}.{extension}" 

235 display_name = _drive_display_name(resume.filename, extension) 

236 try: 

237 stored = store.store_resume(recruiter, job, stored_filename, content, display_name) 

238 except Exception as exc: 

239 raise HTTPException( 

240 status_code=_storage_failure_status_code(), 

241 detail={"code": "RESUME_UPLOAD_FAILED", "message": "Could not store the resume."}, 

242 ) from exc 

243 

244 candidate = Candidate( 

245 job_id=job.job_id, 

246 full_name=identity.full_name, 

247 email=identity.email, 

248 # Adapter-agnostic: a Drive upload returns a drive_file_id, a local 

249 # write doesn't — that distinguishes the store, not settings.app_env. 

250 resume_storage_key=stored.storage_key if stored.drive_file_id is None else None, 

251 resume_drive_file_id=stored.drive_file_id, 

252 resume_drive_url=stored.drive_url, 

253 ) 

254 candidate.form_responses = [ 

255 CandidateFormResponse(field_id=field_id, response_value=value) 

256 for field_id, value in answers.items() 

257 ] 

258 db.add(candidate) 

259 try: 

260 db.commit() 

261 except IntegrityError: 

262 # Race backstop behind the pre-check above: two concurrent 

263 # submissions for the same (job_id, email) both pass the SELECT, only 

264 # one wins the UNIQUE index. 

265 db.rollback() 

266 raise HTTPException(status_code=409, detail=_DUPLICATE_SUBMISSION) from None 

267 db.refresh(candidate) 

268 return candidate

271def list_candidates( 

272 db: Session, 

273 job_id: uuid.UUID, 

274 params: PaginationParams, 

275 submission_status: SubmissionStatus | None, 

276 q: str | None, 

277) -> tuple[list[Candidate], int]: 

278 """Mirrors job_service.list_jobs: a COUNT over the filtered subquery, then 

279 one bounded page. Never touches candidate_form_responses or template_fields 

280 — CandidateOut carries no form_responses — so this stays 1 query 

281 regardless of page size (TC-10).""" 

282 base = select(Candidate).where(Candidate.job_id == job_id, Candidate.deleted_at.is_(None)) 

283 if submission_status is not None: 

284 # Hits ix_candidates_job_status (job_id, submission_status). 

285 base = base.where(Candidate.submission_status == submission_status) 

286 if q: 

287 base = base.where(or_(Candidate.full_name.ilike(f"%{q}%"), Candidate.email.ilike(f"%{q}%"))) 

288 total = db.scalar(select(func.count()).select_from(base.subquery())) or 0 

289 rows = db.scalars( 

290 base.order_by(Candidate.submitted_at.desc(), Candidate.candidate_id.desc()) 

291 .offset((params.page - 1) * params.size) 

292 .limit(params.size) 

293 ).all() 

294 return list(rows), total

297def get_owned_candidate( 

298 db: Session, recruiter_id: uuid.UUID, candidate_id: uuid.UUID 

299) -> Candidate | None: 

300 """The only ownership path for candidates — scoped through the owning job's

304 select(Candidate) 

305 .join(JobPosting, JobPosting.job_id == Candidate.job_id) 

306 .where( 

307 Candidate.candidate_id == candidate_id, 

308 Candidate.deleted_at.is_(None), 

309 JobPosting.recruiter_id == recruiter_id, 

310 JobPosting.deleted_at.is_(None), 

311 ) 

312 )

315def form_responses(db: Session, candidate_id: uuid.UUID) -> list[FormResponseOut]: 

316 """One query, joined to template_fields for the label — never stored on 

317 the response row (AC), and never a per-response lookup (TC-10).""" 

318 rows = db.execute( 

319 select(CandidateFormResponse, TemplateField) 

320 .join(TemplateField, TemplateField.field_id == CandidateFormResponse.field_id) 

321 .where(CandidateFormResponse.candidate_id == candidate_id) 

322 .order_by(TemplateField.field_order) 

323 ).all() 

324 return [ 

325 FormResponseOut( 

326 field_id=field.field_id, 

327 field_label=field.field_label, 

328 field_type=field.field_type, 

329 response_value=response.response_value, 

330 ) 

331 for response, field in rows 

332 ]

344def resolve_resume_path(candidate: Candidate) -> Path | None: 

345 """Resolves resume_storage_key for streaming. Only ever set for a local 

346 write (candidate_service.submit_application only populates it when the 

347 store's StoredFile has no drive_file_id, TS-07), so a Drive-backed 

348 candidate always returns None here and the route 404s, matching 

349 resume_url_for's local-only route. Returns None (never raises) for: no 

350 key stored, file missing, or the resolved path escaping the job's 

351 folder — the route maps all three to 404 uniformly.""" 

352 if not candidate.resume_storage_key: 

353 return None 

354 folder = local_job_folder(candidate.job_id).resolve() 

355 target = Path(candidate.resume_storage_key).resolve() 

356 if folder not in target.parents: 

357 return None 

358 if not target.is_file(): 

359 return None 

360 return target
