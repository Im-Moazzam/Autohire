1"""Resolves which template field answers a candidate's email and full name. 

2 

3`candidates.email` / `candidates.full_name` are real columns, but the form is 

4template-driven — there is no reserved field for either. This module is the 

5single place that maps a field's label to one of those two columns, by a 

6normalised alias match, so template_service (reject at template save time) 

7and candidate_service (read the answer at submission time) can never drift 

8apart on what counts as "the email field". 

9 

10Validating at template save time is the point: a template missing an email 

11field would otherwise only surface as a 500 on some future candidate's 

12submission, for a mistake the recruiter made and the candidate can't fix. 

13""" 

14 

15import re 

16from collections.abc import Sequence 

17 

18 

19def _normalize(label: str) -> str: 

20 return re.sub(r"[^a-z0-9]", "", label.lower()) 

21 

22 

23def resolve_identity_fields(labels: Sequence[str]) -> tuple[int | None, int | None]: 

24 """Returns (email_index, name_index) into `labels` — either may be None. 

25 

26 Operates on plain labels, not field objects, so the caller (a list of 

27 ORM `TemplateField` rows here, `TemplateFieldIn` schemas there — two 

28 different types) does the trivial index-to-object lookup itself instead 

29 of this module needing to be generic over both. 

30 

31 Substring match on the normalised label, not an exact alias table: "Email", 

32 "Your Email Address", and "Work Email" all resolve; only "email" itself is 

33 reserved so a name field never doubles as the email match. First field in 

34 template order wins on a repeat hit. Deliberately permissive — a false 

35 positive here just means the recruiter's 422 fires one field later, a 

36 false negative is the failure mode this check exists to prevent.""" 

37 email_index: int | None = None 

38 name_index: int | None = None 

39 for index, label in enumerate(labels): 

40 norm = _normalize(label) 

41 if email_index is None and "email" in norm: 

42 email_index = index 

43 if name_index is None and "email" not in norm and "name" in norm: 

44 name_index = index 

45 return email_index, name_index
