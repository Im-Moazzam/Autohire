25# Best-effort only: FastAPI has already resolved the `resume: UploadFile` 

26# parameter (and therefore parsed the full multipart body) by the time a 

27# route function runs — see the ponytail note in resume_validation.read_capped 

28# for why that's not worth giving up the declarative UploadFile signature to 

29# fix. This dependency runs before that parsing and rejects the common case 

30# (a client that sends an honest, grossly-oversized Content-Length) without 

31# touching the body at all. A lying header can only make this check miss, 

32# never falsely reject — read_capped is what actually enforces the cap. 

33_CONTENT_LENGTH_SLACK_BYTES = 64 * 1024

36def _reject_oversized_declared_length(request: Request) -> None: 

37 declared = request.headers.get("content-length") 

38 if declared is None: 

39 return 

40 try: 

41 size = int(declared) 

42 except ValueError: 

43 return 

44 if size > settings.max_resume_mb * 1024 * 1024 + _CONTENT_LENGTH_SLACK_BYTES: 

45 raise HTTPException( 

46 status_code=413, 

47 detail={ 

48 "code": "FILE_TOO_LARGE", 

49 "message": f"Resume exceeds {settings.max_resume_mb}MB.", 

50 },

54def _get_public_job(db: Session, apply_slug: str) -> JobPosting: 

55 """Unknown slug, soft-deleted, and DRAFT all 404 identically — a draft is 

56 not "closed", it's unlaunched, and a distinguishable response would leak 

57 that a posting exists before the recruiter has shared it.""" 

58 job = job_service.get_job_by_slug(db, apply_slug) 

59 if job is None or job.status == JobStatus.DRAFT: 

60 raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail=_JOB_NOT_FOUND) 

61 return job
